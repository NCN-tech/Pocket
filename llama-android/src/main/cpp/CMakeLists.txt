cmake_minimum_required(VERSION 3.22.1)
project(pocket_llama)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llamacpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG v0.5.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(llamacpp)
add_library(pocket_llama SHARED pocket_llama.cpp)
target_include_directories(pocket_llama PRIVATE ${llamacpp_SOURCE_DIR}/include ${llamacpp_SOURCE_DIR}/ggml/include)
target_link_libraries(pocket_llama PRIVATE llama android log)
target_compile_features(pocket_llama PRIVATE cxx_std_17)
